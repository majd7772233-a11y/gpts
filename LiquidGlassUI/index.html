/**
 * شاشة تسجيل دخول أندرويد بتصميم Liquid Glass صافي وعميق
 * مع خلفية سائلة متحركة وتفاعلية تستجيب للمس (Interactive Metaballs + Specular Glass)
 */
@Composable
fun LiquidGlassLoginScreen(
    onLoginSuccess: (String, String) -> Unit = { _, _ -> }
) {
    // تفعيل اتجاه اللغة العربية (RTL)
    CompositionLocalProvider(LocalLayoutDirection provides LayoutDirection.Rtl) {
        var email by remember { mutableStateOf("kotlin.dev@android.com") }
        var password by remember { mutableStateOf("123456789") }
        var touchPoint by remember { mutableStateOf(Offset(500f, 900f)) }

        // حركة زمنية مستمرة لتحريك السائل في الخلفية
        val infiniteTransition = rememberInfiniteTransition(label = "liquid_time")
        val time by infiniteTransition.animateFloat(
            initialValue = 0f,
            targetValue = 6.283f, // 2 * PI
            animationSpec = infiniteRepeatable(
                animation = tween(durationMillis = 14000, easing = LinearEasing),
                repeatMode = RepeatMode.Restart
            ),
            label = "time"
        )

        // حركة سلسة تتبع إصبع المستخدم
        val animatedTouch by animateOffsetAsState(
            targetValue = touchPoint,
            animationSpec = spring(dampingRatio = 0.65f, stiffness = Spring.StiffnessLow),
            label = "touch_follower"
        )

        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(Color(0xFF05060C))
                .pointerInput(Unit) {
                    detectDragGestures { change, _ ->
                        touchPoint = change.position
                    }
                }
                .pointerInput(Unit) {
                    detectTapGestures { offset ->
                        touchPoint = offset
                    }
                },
            contentAlignment = Alignment.Center
        ) {
            // 1. الخلفية السائلة التفاعلية (Interactive Liquid Background)
            InteractiveLiquidBackground(
                time = time,
                touchOffset = animatedTouch
            )

            // 2. بطاقة الزجاج السائل الرئيسية (Deep Pure Liquid Glass Card)
            PureLiquidGlassSurface(
                modifier = Modifier
                    .fillMaxWidth(0.88f)
                    .wrapContentHeight(),
                cornerRadius = 34.dp
            ) {
                Column(
                    modifier = Modifier
                        .padding(horizontal = 24.dp, vertical = 32.dp),
                    horizontalAlignment = Alignment.CenterHorizontally
                ) {
                    // أيقونة البصمة داخل قطرة زجاجية بارزة
                    PureLiquidGlassSurface(
                        modifier = Modifier.size(82.dp),
                        cornerRadius = 28.dp
                    ) {
                        Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
                            Icon(
                                imageVector = Icons.Default.Fingerprint,
                                contentDescription = "Fingerprint",
                                tint = Color(0xFF00F2FE),
                                modifier = Modifier.size(44.dp)
                            )
                        }
                    }

                    Spacer(modifier = Modifier.height(16.dp))

                    Text(
                        text = "مرحباً بعودتك",
                        color = Color.White,
                        fontSize = 26.sp,
                        fontWeight = FontWeight.ExtraBold
                    )
                    Text(
                        text = "سجّل الدخول إلى مساحتك الزجاجية",
                        color = Color.White.copy(alpha = 0.65f),
                        fontSize = 13.sp,
                        modifier = Modifier.padding(top = 4.dp)
                    )

                    Spacer(modifier = Modifier.height(28.dp))

                    // حقل البريد الإلكتروني الزجاجي الغائر
                    LiquidRecessedTextField(
                        value = email,
                        onValueChange = { email = it },
                        label = "البريد الإلكتروني",
                        leadingIcon = {
                            Icon(Icons.Default.Email, null, tint = Color(0xFF70E1F5))
                        }
                    )

                    Spacer(modifier = Modifier.height(16.dp))

                    // حقل كلمة المرور الزجاجي الغائر
                    LiquidRecessedTextField(
                        value = password,
                        onValueChange = { password = it },
                        label = "كلمة المرور",
                        isPassword = true,
                        leadingIcon = {
                            Icon(Icons.Default.Lock, null, tint = Color(0xFF70E1F5))
                        }
                    )

                    Spacer(modifier = Modifier.height(28.dp))

                    // زر تسجيل الدخول على شكل قطرة زجاجية بارزة
                    LiquidDropletButton(
                        text = "تسجيل الدخول",
                        onClick = { onLoginSuccess(email, password) }
                    )
                }
            }
        }
    }
}

/**
 * مكوّن الخلفية السائلة التفاعلية
 * يرسم كرات سائلة متوهجة تندمج معاً وتتفاعل مع موقع اللمس
 */
@Composable
fun InteractiveLiquidBackground(
    time: Float,
    touchOffset: Offset
) {
    Canvas(
        modifier = Modifier
            .fillMaxSize()
            .blur(radius = 65.dp, edgeTreatment = BlurredEdgeTreatment.Unbounded)
    ) {
        val w = size.width
        val h = size.height

        // الكرة السائلة 1: سماوي متوهج
        val blob1Center = Offset(
            x = w * 0.3f + cos(time * 2f) * (w * 0.25f),
            y = h * 0.3f + sin(time * 1.5f) * (h * 0.15f)
        )
        drawCircle(
            brush = Brush.radialGradient(
                colors = listOf(Color(0xFF00F2FE), Color(0xFF0072FF), Color.Transparent),
                center = blob1Center,
                radius = w * 0.55f
            ),
            center = blob1Center,
            radius = w * 0.55f
        )

        // الكرة السائلة 2: بنفسجي عميق
        val blob2Center = Offset(
            x = w * 0.7f + sin(time * 1.8f) * (w * 0.22f),
            y = h * 0.68f + cos(time * 2.2f) * (h * 0.18f)
        )
        drawCircle(
            brush = Brush.radialGradient(
                colors = listOf(Color(0xFFB721FF), Color(0xFF21D4FD), Color.Transparent),
                center = blob2Center,
                radius = w * 0.6f
            ),
            center = blob2Center,
            radius = w * 0.6f
        )

        // الكرة السائلة التفاعلية (تتبع إصبع المستخدم)
        drawCircle(
            brush = Brush.radialGradient(
                colors = listOf(
                    Color(0xFF00FFCC).copy(alpha = 0.85f),
                    Color(0xFF00B4DB).copy(alpha = 0.4f),
                    Color.Transparent
                ),
                center = touchOffset,
                radius = w * 0.45f
            ),
            center = touchOffset,
            radius = w * 0.45f
        )
    }
}

/**
 * سطح الزجاج السائل الصافي والعميق (Pure Deep Liquid Glass Surface)
 * يحاكي انكسار الضوء، اللمعان العالي (Specular Rim)، والعمق الداخلي
 */
@Composable
fun PureLiquidGlassSurface(
    modifier: Modifier = Modifier,
    cornerRadius: Dp = 32.dp,
    content: @Composable BoxScope.() -> Unit
) {
    val shape = RoundedCornerShape(cornerRadius)

    Box(
        modifier = modifier
            .clip(shape)
            // تدرج زجاجي شفاف يحاكي انكسار الضوء السطحي
            .background(
                brush = Brush.linearGradient(
                    colors = listOf(
                        Color.White.copy(alpha = 0.14f),
                        Color.White.copy(alpha = 0.03f),
                        Color(0xFF00F2FE).copy(alpha = 0.06f)
                    ),
                    start = Offset.Zero,
                    end = Offset.Infinite
                )
            )
            // رسم الإضاءة الداخلية (Inner Specular Highlight)
            .drawWithContent {
                drawContent()
                // لمعة علوية تحاكي تحدّب السائل
                drawRoundRect(
                    brush = Brush.verticalGradient(
                        colors = listOf(
                            Color.White.copy(alpha = 0.35f),
                            Color.Transparent
                        ),
                        startY = 0f,
                        endY = size.height * 0.25f
                    ),
                    style = androidx.compose.ui.graphics.drawscope.Stroke(width = 2.dp.toPx())
                )
            }
            // إطار زجاجي مضيء من الأعلى واليمين
            .border(
                width = 1.5.dp,
                brush = Brush.linearGradient(
                    colors = listOf(
                        Color.White.copy(alpha = 0.55f),
                        Color.White.copy(alpha = 0.10f),
                        Color(0xFF00F2FE).copy(alpha = 0.35f)
                    )
                ),
                shape = shape
            ),
        content = content
    )
}

/**
 * حقل إدخال زجاجي غائر (Recessed Liquid Input)
 */
@Composable
fun LiquidRecessedTextField(
    value: String,
    onValueChange: (String) -> Unit,
    label: String,
    isPassword: Boolean = false,
    leadingIcon: @Composable (() -> Unit)? = null
) {
    val shape = RoundedCornerShape(20.dp)
    Column(modifier = Modifier.fillMaxWidth()) {
        Text(
            text = label,
            color = Color.White.copy(alpha = 0.8f),
            fontSize = 12.sp,
            modifier = Modifier.padding(bottom = 6.dp, start = 6.dp)
        )
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .clip(shape)
                .background(Color.Black.copy(alpha = 0.32f))
                .border(
                    width = 1.dp,
                    brush = Brush.verticalGradient(
                        colors = listOf(
                            Color.Black.copy(alpha = 0.6f), // ظل علوي غائر
                            Color.White.copy(alpha = 0.25f) // انعكاس سفلي
                        )
                    ),
                    shape = shape
                )
                .padding(horizontal = 16.dp, vertical = 14.dp)
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                if (leadingIcon != null) {
                    leadingIcon()
                    Spacer(modifier = Modifier.width(12.dp))
                }
                BasicTextField(
                    value = value,
                    onValueChange = onValueChange,
                    textStyle = TextStyle(color = Color.White, fontSize = 15.sp),
                    visualTransformation = if (isPassword) PasswordVisualTransformation() else VisualTransformation.None,
                    cursorBrush = SolidColor(Color(0xFF00F2FE)),
                    modifier = Modifier.fillMaxWidth()
                )
            }
        }
    }
}

/**
 * زر قطرة الزجاج السائل البارز (Raised Liquid Droplet Button)
 */
@Composable
fun LiquidDropletButton(
    text: String,
    onClick: () -> Unit
) {
    val shape = RoundedCornerShape(22.dp)
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(54.dp)
            .clip(shape)
            .background(
                brush = Brush.linearGradient(
                    colors = listOf(
                        Color.White.copy(alpha = 0.28f),
                        Color(0xFF00F2FE).copy(alpha = 0.18f),
                        Color.White.copy(alpha = 0.08f)
                    )
                )
            )
            .border(
                width = 1.5.dp,
                brush = Brush.verticalGradient(
                    colors = listOf(
                        Color.White.copy(alpha = 0.75f),
                        Color(0xFF00F2FE).copy(alpha = 0.3f)
                    )
                ),
                shape = shape
            )
            .clickable(onClick = onClick),
        contentAlignment = Alignment.Center
    ) {
        // انعكاس ضوئي علوي يشبه قطرة الماء
        Box(
            modifier = Modifier
                .align(Alignment.TopCenter)
                .padding(top = 4.dp)
                .fillMaxWidth(0.65f)
                .height(10.dp)
                .clip(CircleShape)
                .background(
                    Brush.verticalGradient(
                        listOf(Color.White.copy(alpha = 0.45f), Color.Transparent)
                    )
                )
        )
        Text(
            text = text,
            color = Color.White,
            fontSize = 16.sp,
            fontWeight = FontWeight.Bold
        )
    }
}
